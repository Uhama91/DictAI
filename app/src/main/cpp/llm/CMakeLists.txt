cmake_minimum_required(VERSION 3.22.1)
project(dictai_local_format LANGUAGES C CXX)
if(NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
    message(FATAL_ERROR "LLAMA_SOURCE_DIR must point to the pinned llama.cpp checkout")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_LTO OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
option(DICTAI_ARM82_VARIANT "Build the isolated DOTPROD/FP16 runtime" OFF)
set(DICTAI_LLM_TARGET dictai_llm)
if(DICTAI_ARM82_VARIANT)
    if(NOT ANDROID)
        message(FATAL_ERROR "DICTAI_ARM82_VARIANT is Android-only")
    endif()
    set(DICTAI_LLM_TARGET dictai_llm_arm82)
endif()
if(ANDROID)
    if(DICTAI_ARM82_VARIANT)
        set(GGML_CPU_ARM_ARCH armv8-a+dotprod+fp16 CACHE STRING "Probed ARM64 features" FORCE)
    else()
        set(GGML_CPU_ARM_ARCH armv8-a CACHE STRING "Portable ARM64 baseline" FORCE)
    endif()
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/llama")
add_library(${DICTAI_LLM_TARGET} SHARED local_format_jni.cpp)
target_compile_features(${DICTAI_LLM_TARGET} PRIVATE cxx_std_17)
target_compile_options(${DICTAI_LLM_TARGET} PRIVATE -fvisibility=hidden -fvisibility-inlines-hidden)
if(DICTAI_ARM82_VARIANT)
    target_compile_definitions(${DICTAI_LLM_TARGET} PRIVATE DICTAI_ARM82_VARIANT=1)
endif()
target_link_libraries(${DICTAI_LLM_TARGET} PRIVATE llama)
target_link_options(${DICTAI_LLM_TARGET} PRIVATE
    "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map"
    "-Wl,--exclude-libs,ALL"
    "-Wl,-z,max-page-size=16384")
set_target_properties(${DICTAI_LLM_TARGET} PROPERTIES
    LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
if(NOT ANDROID)
    if(NOT EXISTS "${DICTAI_JAVA_HOME}/include/jni.h")
        message(FATAL_ERROR "DICTAI_JAVA_HOME required for host JNI tests")
    endif()
    target_include_directories(${DICTAI_LLM_TARGET} PRIVATE "${DICTAI_JAVA_HOME}/include" "${DICTAI_JAVA_HOME}/include/linux")
endif()
